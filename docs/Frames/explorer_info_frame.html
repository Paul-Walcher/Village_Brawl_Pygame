391        right_sbox: pygame.Rect = None
392
393        left_tbox, right_tbox = graphics.split_rect_horizontally(rect=self.description_textbox.frame_dim, percentages=[0.5, 0.5])
394        left_sbox, main_sbox, right_sbox = graphics.split_rect_horizontally(rect=self.relative_slider_border_frame, percentages=[0.25, 0.5, 0.25])
395
396        mpos: point_t = graphics.position_relative_point(anchor_point=as_point(self.inner_frame_dim), make_relative_point=event.pos)
397
398        if event.button == Mouse.LEFT_BUTTON:
399

402                self.description_textbox.prev_page()

405            elif calculations.point_collides(mpos, right_tbox):

410            elif calculations.point_collides(mpos, left_sbox):

414
415            elif calculations.point_collides(mpos, right_sbox):

419
420            elif calculations.point_collides(mpos, main_sbox):

422                self.slider_selection_triggered()
423
424        elif event.button == Mouse.WHEEL:
425
426            self.full_image_shown = True
427            self.raise_prerender_flag()
