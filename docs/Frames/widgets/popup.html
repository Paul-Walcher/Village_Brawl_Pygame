774            return

794                                                                                                    )
795            #replacing the highlighted rendered option

797
798        else:
799
800            #shrink the frame, then render the text
801            P: float = (2 * percentage - 1.0)
802            margin: float = (1.0 - P) * self.selected_animation_size_decrease_percentage
803
804            shrunken_frame: pygame.Rect = graphics.apply_margins(
805                                                                    self.options_frame_dim_margins[self.option_index],

808                                                                    )
