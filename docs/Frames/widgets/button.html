285    def prerender(self) -> None:
286        """
287        Rendering without recalculating every position.
288        """
289        fill_color: color_t = (self.selected_background_color if self.selected() else self.background_color)

292        self.button_selected_surface.fill(fill_color)
293
294        if self.background_image_rendered is not None:
295            self.button_not_selected_surface.blit(self.background_image_rendered, (0, 0))
296        if self.button_page_rendered is not None:
297            for surface, position in self.button_page_rendered:
298                self.button_not_selected_surface.blit(surface, position)
299
300        if self.selected_background_image_rendered is not None:
301            self.button_selected_surface.blit(self.selected_background_image_rendered, (0, 0))
302        if self.selected_button_page_rendered is not None:
303            for surface, position in self.selected_button_page_rendered:
304                self.button_selected_surface.blit(surface, position)

315        Makes this buttons state unselected if is currently unselected.

323        Returns whether the press animation is playing or not.

334    def press(self, callback_argument: Any = None) -> bool:
335        """
336        Presses the button and plays the animation.
337        Returns True on success, False if a press is already queued, or a press animation is playing.
338        """
339        if self._press_queued or self.state == Button.PRESSED_ANIMATION_PLAYING:
340            return False
341        else:
342            self._press_queued = True
343            self.callback_argument = callback_argument
344            return True

346    def initialize_pressed_animation(self) -> None:
347        """
348        Initializes the pressed animation.
349        """
350
351        #setting state
352        self.state = Button.PRESSED_ANIMATION_PLAYING
353        #prerendering
354        self.prerender()
355        #copying the current surface
356        self.button_selected_surface_copy = self.button_selected_surface.copy()
357        #starting the timer
358        self.press_animation_clock.start()
359
360    def render_pressed_animation(self) -> None:
361        """
362        Renders the pressed animation.
363        """
364        percentage: float = self.press_animation_clock.elapsed() / self.press_animation_duration
365
366        if percentage >= 1.0:
367            self.finalize_pressed_animation()
368        else:
369
370            if percentage <= 0.5:
371
372                P = percentage * 2
373                new_size_p: float = 1.0 - (P * self.press_animation_size_decrease)
374
375                self.button_selected_surface = graphics.scale_surface(
376                                                                        self.button_selected_surface_copy,
377                                                                        scale=new_size_p,
378                                                                        scale_function=pygame.transform.smoothscale
379                                                                        )
380            else:
381
382                P = (percentage - 0.5) * 2
383                new_size_p: float = (1.0 - self.press_animation_size_decrease) + (P * self.press_animation_size_decrease)
384
385                self.button_selected_surface = graphics.scale_surface(
386                                                                        self.button_selected_surface_copy,
387                                                                        scale=new_size_p,
388                                                                        scale_function=pygame.transform.smoothscale
389                                                                        )
390
391    def finalize_pressed_animation(self) -> None:
392        """
393        Finalizes the press animation, then executes self.callback_func.
394        """
395
396        callback_result: Any = None
397        #calling the callback function
398        if self.callback_func is not None:
399            callback_result = self.callback_func(self.callback_argument)
400        #resetting the callback argument
401        self.callback_argument = None
402        #adding the button press finished event
403        self.push_event(Button.ButtonPressFinishedEvent(callback_result=callback_result))
404        self.state = Button.SELECTED
405
406        self.button_selected_surface = self.button_selected_surface_copy
407        self.button_selected_surface_copy = None
408        self.prerender()

411    def tick(self) -> Frame:
412        """
413        Renders the animation if it is playing.
414        """
415
416        #checking for queued input
417        if self._press_queued:
418            self._press_queued = False
419            self.initialize_pressed_animation()
420
421        if self.state == Button.PRESSED_ANIMATION_PLAYING:
422            self.render_pressed_animation()
423
424        return self

427
428        if self.selected():
429            #surface offset
430            x_off, y_off = (self.width - self.button_selected_surface.get_width()) // 2, (self.height - self.button_selected_surface.get_height()) // 2
431            screen.blit(self.button_selected_surface, (self.x + x_off, self.y + y_off))
432        else:
433            x_off, y_off = (self.width - self.button_not_selected_surface.get_width()) // 2, (self.height - self.button_not_selected_surface.get_height()) // 2
434            screen.blit(self.button_not_selected_surface, (self.x + x_off, self.y + y_off))
