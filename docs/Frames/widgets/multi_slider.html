569    def tick(self) -> Frame:
570        """
571        Slider tick method.
572        """
573
574        for slider in self.sliders:
575            slider.tick()
576
577        if self.state == MultiSlider.NOT_SLIDING:
578
579            #check for events
580            if len(self._assembled_events) > 0:
581                #start new animation
582                event_ref: Event = self._assembled_events[0]
583
584                if event_ref.type == MultiSlider.EventTypes.SELECT_ANIMATION_ENDED:
585
586                    #resetting await flags
587                    self._reset_awaiting_list(indices=event_ref.sliders_active_indices)
588                    #starting the select animation
589                    for slider in event_ref.sliders_active:
590                        slider.select()
591
592                    self.state = MultiSlider.SELECT_ANIMATION_PLAYING
593
594                    self.select_animation_queued = False
595
596                elif event_ref.type == MultiSlider.EventTypes.FLIPPING_ANIMATION_ENDED:
597
598                    #resetting await flags
599                    self._reset_awaiting_list(indices=event_ref.sliders_active_indices)
600
601                    for i in range(len(event_ref.sliders_flipped_all)):
602
603                        flip_flag: bool = event_ref.sliders_flipped_all[i]
604
605                        if flip_flag is not None:
606                            if flip_flag:
607                                self.sliders[i].flip_all_cards()
608                            else:
609                                self.sliders[i].flip_card()
610
611
612                    self.state = MultiSlider.FLIPPING_ANIMATION_PLAYING
613
614                    self.flipping_animation_queued = False
615
616                elif event_ref.type == MultiSlider.EventTypes.SLIDING_ANIMATION_ENDED:
617
618                    #resetting await flags
619                    self._reset_awaiting_list(indices=event_ref.sliders_active_indices)
620
621                    for i in range(len(event_ref.slide_directions)):
622
623                        slide_dir: bool = event_ref.slide_directions[i]
624
625                        if slide_dir is not None:
626                            if slide_dir:
627                                self.sliders[i].slide_right()
628                            else:
629                                self.sliders[i].slide_left()
630
631
632                    self.state = MultiSlider.SLIDING
633
634                    self.slides -= 1
635
636        elif self.state == MultiSlider.SELECT_ANIMATION_PLAYING:
637
638            if self._awaiting_finished():
639                #select animation ended
640                event_ref: MultiSlider.SelectAnimationEndedEvent = self._assembled_events[0]
641                event_ref.selected_indices = self.get_selected_index()
642
643                self.push_event(event_ref)
644                self._assembled_events.pop(0)
645
646                self.state = MultiSlider.NOT_SLIDING
647
648        elif self.state == MultiSlider.FLIPPING_ANIMATION_PLAYING:
649
650            if self._awaiting_finished():
651                #flipping animation ended
652                event_ref: MultiSlider.FlippingAnimationEndedEvent = self._assembled_events[0]
653                event_ref.selected_indices = self.get_selected_index()
654
655                self.push_event(event_ref)
656                self._assembled_events.pop(0)
657
658                self.state = MultiSlider.NOT_SLIDING
659
660        elif self.state == MultiSlider.SLIDING:
661
662            if self._awaiting_finished():
663                #flipping animation ended
664                event_ref: MultiSlider.SlidingAnimationEndedEvent = self._assembled_events[0]
665                event_ref.selected_indices = self.get_selected_index()
666
667                self.push_event(event_ref)
668                self._assembled_events.pop(0)
669
670                self.state = MultiSlider.NOT_SLIDING
671
672        #polling events
673        for i in range(len(self.sliders)):
674
675            slider: Slider = self.sliders[i]
676
677            for event in slider.poll_events():
678
679                if event.type == Slider.EventTypes.SELECT_ANIMATION_ENDED:
680                    #register the animation ended
681                    self._awaiting_list[i] = True
682
683                elif event.type == Slider.EventTypes.FLIPPING_ANIMATION_ENDED:
684                    #register the animation ended
685                    self._awaiting_list[i] = True
686
687                elif event.type == Slider.EventTypes.SLIDING_ANIMATION_ENDED:
688                    #register the animation ended
689                    self._awaiting_list[i] = True
690
691        if (self.state != MultiSlider.NOT_SLIDING):
692            self.raise_prerender_flag()
693
694        if self.prerender_flag_raised():
695            self.prerender()
696            self.reset_prerender_flag()
697
698        return self
