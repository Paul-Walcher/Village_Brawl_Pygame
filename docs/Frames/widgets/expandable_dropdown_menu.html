171                                    options_dict: expandable_tuple_options_t,
172                                    text_color: color_t,
173                                    highlighted_text_color: color_t,
174
175                                    prev_tuple: tuple[str] = None #the previous tuple, if its a recursive call
176
177                                    ) -> tuple[expandable_dropdown_options_t, dict[ tuple[ tuple[str] ] , DropDownKey]]:
178        """
179        Given a dictionary with keys like:
180
181        options = {
182                    ("line 1", "line 2"): {
183                                                ("E",): None,
184                                                ("RT", "TZ", "IO"): {
185                                                                        ("ERERER", "BZTT"): None,
186                                                                        ("HBEDZG",): None,
187                                                                        ("EE", "RR"): None,
188                                                                        ("WWW", "TT"): None,
189
190                                                                        "n_options_shown": 2
191                                                                            }
192                                            },
193                    ("LINE",): None,
194                    ("JJ", "JJ", "JJ", "JJ"): None,
195                    ("GGG",): None,
196                    ("JKIJI", "IJUHU"): None,
197                    "n_options_shown": 3
198
199                    }
200        This function returns the same dictionary, but with the tuple keys
201        transformed into (colored_page, highlighted_colored_page) tuples.
202
203        The second argument that is returned is a mapping of tuples: dropdownkey.
204
205        E.g. in the returned dict for options the key
206        (("line 1", "line 2"), ("E",)) -> Dropdownkey for ("E",)
207        """
208
209        back_dict: expandable_dropdown_options_t = {}
210        mapping_dict: dict[ tuple[tuple[str]], DropDownKey ] = {}
211
212        for key, value in options_dict.items():
213
214            if type(key) == tuple:
215
216                #transforming the key
217                option_page: colored_page_t = graphics.page_to_colored_page(list(key), text_color)
218                highlighted_option_page: colored_page_t = graphics.page_to_colored_page(list(key), highlighted_text_color)
219
220                new_options: expandable_dropdown_options_t = None
221                new_mappings: dict[ tuple[ tuple[str] ] , DropDownKey] = None
222
223                #mapping dict
224                tuple_key: tuple[tuple[str]] = (key,)
225                if prev_tuple is not None:
226                    tuple_key = (*prev_tuple, key)
227
228                if type(value) == dict:
229                    new_options, new_mappings = ExpandableDropdownMenu.page_keys_to_dropdownkey(
230                                                                                                options_dict=value,
231                                                                                                text_color=text_color,
232                                                                                                highlighted_text_color=highlighted_text_color,
233
234                                                                                                prev_tuple=tuple_key
235                                                                                                )
236                    mapping_dict.update(new_mappings)
237
238                new_key = DropDownKey(option_page=option_page, highlighted_option_page=highlighted_option_page)
239
240                back_dict[new_key] = new_options
241                mapping_dict[tuple_key] = new_key
242
243            else:
244                back_dict[key] = value
245
246        return (back_dict, mapping_dict)

347    def _calculate_depth_and_max_length(self, options: expandable_dropdown_options_t) -> tuple[int, int]:
348        """
349        returns the depth of the deepest nested dropdown menu
350        we use a deque in order to avoid putting to many calls on the stack
351        """
352        mdepth = deque([(options, 1)])
353        max_depth = 1
354        max_length = -1
355
356        #as long as we havent processed each option
357        while len(mdepth) > 0:
358            #getting the current option and depth
359            option, depth = mdepth.popleft()
360            #getting the amount of options in the dropdown
361            max_length = max(max_length, len(list(option.values())))
362            for v in option.values():
363                if v is not None:
364                    #if any of the options values is not None, that means its a dropdown menu
365                    max_depth = max(max_depth, depth+1)#we need to check max deptg
366                    mdepth.append((v, depth+1)) #add the next option to processing
367
368        return max_depth, max_length

537        """

351        we use a deque in order to avoid putting to many calls on the stack

355        max_length = -1

358        while len(mdepth) > 0:
359            #getting the current option and depth
360            option, depth = mdepth.popleft()
361            #getting the amount of options in the dropdown
362            max_length = max(max_length, len(list(option.values())))
363            for v in option.values():
364                if v is not None:
365                    #if any of the options values is not None, that means its a dropdown menu
366                    max_depth = max(max_depth, depth+1)#we need to check max deptg
367                    mdepth.append((v, depth+1)) #add the next option to processing
368
