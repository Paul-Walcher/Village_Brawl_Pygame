162    def prerender(self):

179        for i in range(self.n_options):
180            if (i == self.selected_index):

183            else:

241    def render_select_animation(self):
242        #calculates the select animation
243
244        #what percentage elapsed of the animation duration
245        percentage = self.select_clock.elapsed() / self.selected_animation_duration
246        #if the percenatge >= 1.0, finish the animation and selection process
247        if percentage >= 1.0:
248            self.finalize_select_animation()
249            return
250
251        else:
252
253            #The animation works by first decreasing the textsize, then increasing it
254            if percentage <= 0.5:
255
256                fp = 2*percentage
257
258                cx = self.inner_frame_dim.x - self.x
259                cy = self.inner_frame_dim.y - self.y + (self.selected_index) * self.full_box_height
260                cw = self.option_box_width
261                ch = self.option_box_height
262
263                mx, my = self.text_margins_percentage
264                hmx, hmy = self.highlighted_text_margins_percentage
265
266                mxd, myd = hmx + (mx - hmx) * fp, hmy + (my - hmy) * fp
267
268
269                sframe = pygame.Rect(cx, cy, cw, ch)
270                rframe = graphics.apply_margins(sframe, mxd, myd)
271
272                rtext = graphics.render_page_from_chars(self.font, self.options[self.selected_index], rframe,
273                                                        text_color=self.highlighted_text_color, centered=True,
274                                                        vertical_centered=True)
275
276                self.rendered_highlighted_texts[self.selected_index] = rtext
277
278            else:
279
280                fp = 2 * (percentage - 0.5)
281
282                cx = self.inner_frame_dim.x - self.x
283                cy = self.inner_frame_dim.y - self.y + (self.selected_index) * self.full_box_height
284                cw = self.option_box_width
285                ch = self.option_box_height
286
287                mx, my = self.text_margins_percentage
288                hmx, hmy = self.highlighted_text_margins_percentage
289
290                mxd, myd = mx + (hmx - mx) * fp, my + (hmy - my) * fp
291
292                sframe = pygame.Rect(cx, cy, cw, ch)
293                rframe = graphics.apply_margins(sframe, mxd, myd)
294
295                rtext = graphics.render_page_from_chars(self.font, self.options[self.selected_index], rframe,
296                                                        text_color=self.highlighted_text_color, centered=True,
297                                                        vertical_centered=True)
298
299                self.rendered_highlighted_texts[self.selected_index] = rtext

323
324        #move then render
325        if self.state in self.move_states:
326            self.selected_index += self.moves_buffered
327            self.selected_index %= self.n_options
328            self.moves_buffered = 0

332            self.render_select_animation()
