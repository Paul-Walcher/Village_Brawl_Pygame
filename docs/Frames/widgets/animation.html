181    def _initialize_animation(
182                                self,
183                                starting_index: int = 0, #index of the surface to start
184                                shift: int = 0, #shift of the animation
185                                n_loops: int = 1, #how many times the animation loops
186                                new_surface_blit_durations: list[float] = None #the new surface_blit_durations
187                                ) -> None:
188        """
189        Initializes all variables before an animation.
190        """
191        self.surface_index = starting_index + shift
192        self.shift = shift
193        self.n_loops = n_loops
194        self.elapsed_time = 0
195
196        self.looping_index = 0
197
198        if new_surface_blit_durations is not None:
199            self.surface_blit_durations = new_surface_blit_durations
200            self._check_surface_blit_durations()
201
202        #calculating the accumulated blit durations
203        self.surface_blit_durations_accumulated = [self.surface_blit_durations[shift]]
204        for i in range(shift+1, len(self.surface_blit_durations) + shift):
205            self.surface_blit_durations_accumulated.append(self.surface_blit_durations_accumulated[(i-1) % len(self.surface_blit_durations)]\
206                                                            + self.surface_blit_durations[i % len(self.surface_blit_durations)])
207        #now scaling the surfaces
208        for i in range(len(self.surfaces)):
209
210            surf: pygame.Surface = self.surfaces[i]
211
212            if (surf.get_width() != self.width or surf.get_height() != self.height):
213                surf = self.scale_function(surf, (self.width, self.height))
214                self.surfaces[i] = surf
215
216        self.animation_finished = False

220                        self,
221                        starting_index: int = -1,
222                        shift: int = -1,
223                        n_loops: int = -1,
224                        new_surface_blit_durations: list[float] = None

226        """

231            or shift != -1\
232            or n_loops != -1\
233            or new_surface_blit_durations is not None
234            ):
235
236            starting_index = (starting_index if starting_index != -1 else 0)
237            shift = (shift if shift != -1 else 0)
238            n_loops = (n_loops if n_loops != -1 else 1)
239
240            self._initialize_animation(
241                                        starting_index=starting_index,
242                                        shift=shift,
243                                        n_loops=n_loops,
244                                        new_surface_blit_durations=new_surface_blit_durations
245                                        )
246
247            self.push_event(Animation.AnimationStartedEvent(self))
248
249        else:
250
251            self.push_event(Animation.AnimationRestartedEvent(self))
252
253        self.is_paused = False
254        self.animation_clock.start()

316    @override
317    def tick(self) -> None:
318        """
319        The tick function.
320        """
321
322        #renders the tick function
323        if not self.is_paused and not self.animation_finished:
324
325            #rendering
326            self.elapsed_time = self.animation_clock.elapsed()
327
328            if self.elapsed_time >= self.surface_blit_durations_accumulated[self.surface_index]:
329                #next surface is the target
330                self.surface_index += 1
331                self.surface_index %= len(self.surfaces)
332
333                if self.surface_index == self.shift:
334                    #new animation, or stopping
335
336                    if self.looping_index < (self.n_loops-1):
337                        self.looping_index += 1
338                        self.elapsed_time = 0
339                        self.push_event(Animation.AnimationLoopedEvent(self))
340                        self.animation_clock.start()
341                    else:
342                        #stop the animation
343                        self.is_paused = True
344                        self.animation_finished = True
345                        self.push_event(Animation.AnimationEndedEvent(self))
