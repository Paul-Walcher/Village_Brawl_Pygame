129    @override
130    def get_fixed_image_positions(self) -> list[tuple[int, int]]:
131        """
132        Creates the fixed image positions.
133        """
134        surface_distance_covered: int = self.surface_distance * 2 * self.span
135        remaining_space: int = (
136                                self.width - surface_distance_covered if self.horizontal\
137                                else self.height - surface_distance_covered
138                                )
139
140        space_split: int = remaining_space // (1 + 2 * self.span)
141
142        dspaces: list[tuple[int, int]] = []
143        #list of position tuples: (start, length)
144        cx: int = 0
145        """
146        Start position.
147        """
148        for i in range(1 + 2 * self.span):
149
150            dspaces.append((cx, space_split))
151            cx += space_split + self.surface_distance
152
153        return dspaces
