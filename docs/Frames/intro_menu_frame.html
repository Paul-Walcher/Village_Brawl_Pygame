351        Callback
352        """

357            self.state = IntroMenuFrame.MAIN_OPTIONS_FOCUSED

359    def up_pressed(self, event, data) -> None:
360        """
361        Callback
362        """
363
364        if self.state == IntroMenuFrame.NOTHING_FOCUSED:
365            self.state = IntroMenuFrame.MAIN_OPTIONS_FOCUSED

382    def enter_pressed(self, event, data) -> None:
383        """

394    def tick(self) -> None:
395        """
396        Overwritten from Frame class.
397        """
398        if self.intro_delay_clock.elapsed() > self.intro_delay:
399            self.manage_pygame_events()
400
401        if self.exit_flag_raised():
402            if self._next_frame is not None:
403                self._next_frame.focus()
404            return self._next_frame
405
406        return self

408    def render(self, screen: pygame.Surface) -> None:
409        """
410        Overwritten from Frame class.
411        """
412
413        screen.fill(Colors.BLACK)
414
415        renderables: list[rendered_surface_position_tuples_t] = []
416
417        if self.state == IntroMenuFrame.BACK_OPTION_FOCUSED:
418            renderables.append(self.selected_back_text_rendered)
419        else:
420            renderables.append(self.back_text_rendered)
421
422        for i in range(len(self.text_options)):
423
424            if self.state == IntroMenuFrame.MAIN_OPTIONS_FOCUSED and i == self.text_options_index:
425                renderables.append(self.selected_options_rendered[i])
426            else:
427                renderables.append(self.options_rendered[i])
428
429        for rspt in renderables:
430            for surface, position in rspt:
431                screen.blit(surface, position)
