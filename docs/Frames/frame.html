 18    """
 19    This class is the basic framework for a shown window instance.
 20
 21    It has an internal rectangle defined as frame_dim, which
 22    specifies the rectangle in which it operates in (its window size).
 23
 24    The internal rectangles x, y, width and height are separately stored as
 25    class varriables.
 26
 27    IMPORTANT: Do not change a frames frame_dim by doing frame_dim.x = ...,
 28    because internal x, y, width and height and frame_dim are adjusted using setter
 29    methods.
 30
 31    Each frame can also have a parent frame. If this class is discontinued using raise_exit_flag(),
 32    and the parent frames focus group is the same as this classes', the parent frame will receive the focus.
 33
 34    About focus groups:
 35
 36    A focus group is an identifier of the group the frame belongs to.
 37    Inside this group, only one frame can have the focus at one time.
 38
 39    You can check if you have the focus using Frame.is_focused,
 40    and focus as well as unfocus using Frame.focus and Frame.unfocus().
 41
 42    If you dont pass an argument for the focus group to the constructor,
 43    the frame will be assigned to a standard focus group.
 44    You should bind input received for the frame using bind_events().
 45    Whenever focus changes for an event, all event_callbacks are cleared,
 46    and Frame.bind_events is called. There you can bind callbacks depending on whether
 47    you have the focus or not.
 48
 49    A frame handles pygame events by using Frame.manage_pygame_events in Frame.tick.
 50    You can register event callbacks using Frame.add_pygame_key_callback or
 51    Frame.add_pygame_event_callback.
 52    """

 68    class EventCallback(Event):
 69
 70        """
 71        Class for a callback event.
 72
 73        This class is callable, and executes the function
 74        with the data given to the constructor.
 75
 76        The callback function receives two arguments:
 77        event: the pygame.event.Event reference,
 78        data: the data given to this class in the constructor.
 79
 80        The callback function should return 1 of the following 3 enums:
 81
 82        EventCallback.EXIT: signals to the frame processing the callback event
 83                            to call raise_exit_flag(), and this way end the frame's activity.
 84                            This will lead to the event being consumed.
 85
 86        EventCallback.FAILURE:  Signals that the callback function was not able to
 87                                handle the event. This leads to the event not being consumed.
 88
 89        EventCallback.SUCCESS:  Signals the event was successfully handled.
 90                                This will lead to the event being consumed.
 91
 92        if EventCallback.putback == True, the event will be not consumed by this callback
 93        regardless of the return value
 94        """
 95
 96        EXIT = R_ID()
 97        FAILURE = R_ID()
 98        SUCCESS = R_ID()

281    def __prune_parent_frames(self) -> None:
282        """
283        This constructs the parent frame list.
284        Then if the list is bigger that Frame.max_parent_frames,
285        the oldest frames will be cutoff and cleaned up
286        by the garbage collector.
287        """
288        if self.parent_frame is not None:
289            self._parent_frame_list.insert(0, self.parent_frame)
290            self._parent_frame_list.extend(self.parent_frame._parent_frame_list)
291
292        if len(self._parent_frame_list) > self.max_parent_frames:
293            self._parent_frame_list = self.__cutoff_parent_frames(self, self.max_parent_frames)
294
295    def __cutoff_parent_frames(self, cframe: Frame, pindex: int) -> list[Frame]:
296        """
297        Cuts off the oldest frames, and removes them from memory.
298        Returns the modified _parent_frame_list for the frame that called
299        this function to adopt.
300        """
301        if cframe.parent_frame is None:
302            return []
303        if pindex == 0:
304            cframe.parent_frame = None
305            cframe._parent_frame_list = []
306            return cframe._parent_frame_list.copy()
307        else:
308            back_list = self.__cutoff_parent_frames(cframe.parent_frame, pindex-1)
309            cframe._parent_frame_list = [cframe.parent_frame] + back_list
310
311            return cframe._parent_frame_list.copy()

402        """
403        Takes the focus from this frame.
404        If this frame had the focus, and the parent frame has the same focus group as this frame,
405        the parent frame will receive the focus via parent_frame.focus().
406        """
407        if self.focus_group is not None and self.focus_group in Frame.FOCUS_GROUPS and Frame.FOCUS_GROUPS[self.focus_group] == self:

409            Frame.FOCUS_GROUPS[self.focus_group] = None

455    def manage_pygame_events(self) -> None:
456        """
457        Manages the events, by going through all
458        events in the pygame.event.Event queue.
459        Should be called in tick().
460        triggers the event_callbacks in self.__event_callbacks
461        added via self.add_pygame_key_callback or self.add_pygame_event_callback
462        if the callback has putback == True, the event stays in the queue
463        for the next event check
464        """
465
466        putback_events = []
467
468        for event in pygame.event.Event.get():
469            processed = True
470            matched_any = False
471
472            if event.type == pygame.KEYDOWN:
473
474                if event.key not in self.__key_scancodes:
475                    self.__key_scancodes[event.key] = True
476
477                self.__key_scancodes[event.key] = True
478
479            if event.type == pygame.KEYUP:
480
481                if event.key not in self.__key_scancodes:
482                    self.__key_scancodes[event.key] = False
483
484                self.__key_scancodes[event.key] = False
485
486            for ecallback in self.__event_callbacks:
487
488                # First, always match the event type.
489                if event.type != ecallback.type:
490                    continue
491
492                # Then, filter key-specific callbacks.
493                if event.type == pygame.KEYDOWN:
494
495                    if (
496                        ecallback.data is not None
497                        and "key" in ecallback.data
498                        and event.key != ecallback.data["key"]
499                    ):
500                        continue
501
502                matched_any = True
503
504                res = ecallback(event)
505
506                if res == Frame.EventCallback.EXIT:
507                    self.raise_exit_flag()
508
509                if res == Frame.EventCallback.FAILURE:
510                    processed = False
511
512                if ecallback.putback:
513                    processed = False
514
515            if not processed or not matched_any:
516                putback_events.append(event)
517
518        for event in putback_events:
519            pygame.event.Event.post(event)

588    def __key_callback_wrapper( self,
589                                event: pygame.event.Event,
590                                key: int,
591                                data: Any,
592                                putback: bool,
593                                func: event_callback_t,
594                                key_delay: float
595                                ) -> event_callback_result_t:
596
597        """
598        Wrapper function for a event callback added using
599        self.add_pygame_key_callback().
600        """
601
602        # This callback should only receive matching KEYDOWN events.
603        if event.type != pygame.KEYDOWN or event.key != key:
604            return Frame.EventCallback.FAILURE
605
606        if self.pygame_key_is_disabled(key):
607            #if the key is disabled, return SUCCESS for event consumption.
608            return Frame.EventCallback.SUCCESS
609
610        if data["clock"].elapsed() <= key_delay:
611            #if not enough time elapsed for another keypress to be registered,
612            #consume the event with a SUCCESS notification.
613            return Frame.EventCallback.SUCCESS
614
615        back = func(event, data)
616        data["clock"].start()
617
618        if back is None:
619            return Frame.EventCallback.SUCCESS
620
621        return back

623    def add_pygame_key_callback(self, key: int, data: Any, putback: bool, func: event_callback_t, key_delay: float = 100) -> Frame.EventCallback:
624        """
625        Adds a callback for a pygame.KEYDOWN event with event.key == key.
626        Returns the created EventCallback.
627
628        key_delay is the time in milliseconds that needs to pass before this callback can be
629        called again.
630        """
631        if data is None:
632            data = {}
633        else:
634            data = data.copy()
635        data["clock"] = Clock()
636        data["clock"].start()
637        data["key"] = key
638
639        cb = Frame.EventCallback(pygame.KEYDOWN, data, putback, lambda event, d: self.__key_callback_wrapper(event, key, d, putback, func, key_delay))
640        self.__event_callbacks.append(cb)
641        return cb

 19    """
 20    This class is the basic framework for a shown window instance.
 21
 22    It has an internal rectangle defined as frame_dim, which
 23    specifies the rectangle in which it operates in (its window size).
 24
 25    The internal rectangles x, y, width and height are separately stored as
 26    class varriables.
 27
 28    IMPORTANT: Do not change a frames frame_dim by doing frame_dim.x = ...,
 29    because internal x, y, width and height and frame_dim are adjusted using setter
 30    methods.
 31
 32    Each frame can also have a parent frame. If this class is discontinued using raise_exit_flag(),
 33    and the parent frames focus group is the same as this classes', the parent frame will receive the focus.
 34
 35    About focus groups:
 36
 37    A focus group is an identifier of the group the frame belongs to.
 38    Inside this group, only one frame can have the focus at one time.
 39

 69    class EventCallback(Event):
 70
 71        """
 72        Class for a callback event.
 73
 74        This class is callable, and executes the function
 75        with the data given to the constructor.
 76
 77        The callback function receives two arguments:
 78        event: the pygame.event.Event reference,
 79        data: the data given to this class in the constructor.
 80
 81        The callback function should return 1 of the following 3 enums:
 82
 83        EventCallback.EXIT: signals to the frame processing the callback event
 84                            to call raise_exit_flag(), and this way end the frame's activity.
 85                            This will lead to the event being consumed.
 86
 87        EventCallback.FAILURE:  Signals that the callback function was not able to
 88                                handle the event. This leads to the event not being consumed.
 89
 90        EventCallback.SUCCESS:  Signals the event was successfully handled.
 91                                This will lead to the event being consumed.
 92
 93        if EventCallback.putback == True, the event will be not consumed by this callback
 94        regardless of the return value
 95        """
 96
 97        EXIT = R_ID()
 98        FAILURE = R_ID()
 99        SUCCESS = R_ID()
100
101        def __init__(self, type: int, data: Any, putback: bool, func: event_callback_t):
102
103            self.putback: bool = putback
104            """
105            If putback == True, this event callback won't consume the event.
106            """
107            self.func: event_callback_t = func
108            """
109            The function executed when this class is called.
110            """
111            super().__init__(type, data)
112
113        def __call__(self, event: pygame.event.Event) -> event_callback_result_t:
114            """
115            Executes the callback function with the event and this classes' data.
116            """
117            return self.func(event, self.data)

282    def __prune_parent_frames(self) -> None:
283        """
284        This constructs the parent frame list.
285        Then if the list is bigger that Frame.max_parent_frames,
286        the oldest frames will be cutoff and cleaned up
287        by the garbage collector.
288        """
289        if self.parent_frame is not None:
290            self._parent_frame_list.insert(0, self.parent_frame)
291            self._parent_frame_list.extend(self.parent_frame._parent_frame_list)
292
293        if len(self._parent_frame_list) > self.max_parent_frames:
294            self._parent_frame_list = self.__cutoff_parent_frames(self, self.max_parent_frames)
295
296    def __cutoff_parent_frames(self, cframe: Frame, pindex: int) -> list[Frame]:
297        """
298        Cuts off the oldest frames, and removes them from memory.
299        Returns the modified _parent_frame_list for the frame that called
300        this function to adopt.
301        """
302        if cframe.parent_frame is None:
303            return []
304        if pindex == 0:
305            cframe.parent_frame = None
306            cframe._parent_frame_list = []
307            return cframe._parent_frame_list.copy()
308        else:
309            back_list = self.__cutoff_parent_frames(cframe.parent_frame, pindex-1)
310            cframe._parent_frame_list = [cframe.parent_frame] + back_list
311
312            return cframe._parent_frame_list.copy()

456    def manage_pygame_events(self) -> None:
457        """
458        Manages the events, by going through all
459        events in the pygame.event.Event queue.
460        Should be called in tick().
461        triggers the event_callbacks in self.__event_callbacks
462        added via self.add_pygame_key_callback or self.add_pygame_event_callback
463        if the callback has putback == True, the event stays in the queue
464        for the next event check
465        """
466
467        putback_events = []
468
469        for event in pygame.event.Event.get():
470            processed = True
471            matched_any = False
472
473            if event.type == pygame.KEYDOWN:
474
475                if event.key not in self.__key_scancodes:
476                    self.__key_scancodes[event.key] = True
477
478                self.__key_scancodes[event.key] = True
479
480            if event.type == pygame.KEYUP:
481
482                if event.key not in self.__key_scancodes:
483                    self.__key_scancodes[event.key] = False
484
485                self.__key_scancodes[event.key] = False
486
487            for ecallback in self.__event_callbacks:
488
489                # First, always match the event type.
490                if event.type != ecallback.type:
491                    continue
492
493                # Then, filter key-specific callbacks.
494                if event.type == pygame.KEYDOWN:
495
496                    if (
497                        ecallback.data is not None
498                        and "key" in ecallback.data
499                        and event.key != ecallback.data["key"]
500                    ):
501                        continue
502
503                matched_any = True
504
505                res = ecallback(event)
506
507                if res == Frame.EventCallback.EXIT:
508                    self.raise_exit_flag()
509
510                if res == Frame.EventCallback.FAILURE:
511                    processed = False
512
513                if ecallback.putback:
514                    processed = False
515
516            if not processed or not matched_any:
517                putback_events.append(event)
518
519        for event in putback_events:
520            pygame.event.Event.post(event)

589    def __key_callback_wrapper( self,
590                                event: pygame.event.Event,
591                                key: int,
592                                data: Any,
593                                putback: bool,
594                                func: event_callback_t,
595                                key_delay: float
596                                ) -> event_callback_result_t:
597
598        """
599        Wrapper function for a event callback added using
600        self.add_pygame_key_callback().
601        """
602
603        # This callback should only receive matching KEYDOWN events.
604        if event.type != pygame.KEYDOWN or event.key != key:
605            return Frame.EventCallback.FAILURE
606
607        if self.pygame_key_is_disabled(key):
608            #if the key is disabled, return SUCCESS for event consumption.
609            return Frame.EventCallback.SUCCESS
610
611        if data["clock"].elapsed() <= key_delay:
612            #if not enough time elapsed for another keypress to be registered,
613            #consume the event with a SUCCESS notification.
614            return Frame.EventCallback.SUCCESS
615
616        back = func(event, data)
617        data["clock"].start()
618
619        if back is None:
620            return Frame.EventCallback.SUCCESS
621
622        return back

627        Returns the created EventCallback.
628
629        key_delay is the time in milliseconds that needs to pass before this callback can be
630        called again.
631        """
632        if data is None:
633            data = {}
634        else:
635            data = data.copy()
636        data["clock"] = Clock()
637        data["clock"].start()
638        data["key"] = key
639
640        cb = Frame.EventCallback(pygame.KEYDOWN, data, putback, lambda event, d: self.__key_callback_wrapper(event, key, d, putback, func, key_delay))
641        self.__event_callbacks.append(cb)
642        return cb

This class is the basic framework for a shown window instance.

It has an internal rectangle defined as frame_dim, which specifies the rectangle in which it operates in (its window size).

The internal rectangles x, y, width and height are separately stored as class varriables.

IMPORTANT: Do not change a frames frame_dim by doing frame_dim.x = ..., because internal x, y, width and height and frame_dim are adjusted using setter methods.

Each frame can also have a parent frame. If this class is discontinued using raise_exit_flag(), and the parent frames focus group is the same as this classes', the parent frame will receive the focus.

About focus groups:

A focus group is an identifier of the group the frame belongs to. Inside this group, only one frame can have the focus at one time.

You can check if you have the focus using Frame.is_focused, and focus as well as unfocus using Frame.focus and Frame.unfocus().

If you dont pass an argument for the focus group to the constructor, the frame will be assigned to a standard focus group. You should bind input received for the frame using bind_events(). Whenever focus changes for an event, all event_callbacks are cleared, and Frame.bind_events is called. There you can bind callbacks depending on whether you have the focus or not.

A frame handles pygame events by using Frame.manage_pygame_events in Frame.tick. You can register event callbacks using Frame.add_pygame_key_callback or Frame.add_pygame_event_callback.

456    def manage_pygame_events(self) -> None:
457        """
458        Manages the events, by going through all
459        events in the pygame.event.Event queue.
460        Should be called in tick().
461        triggers the event_callbacks in self.__event_callbacks
462        added via self.add_pygame_key_callback or self.add_pygame_event_callback
463        if the callback has putback == True, the event stays in the queue
464        for the next event check
465        """
466
467        putback_events = []
468
469        for event in pygame.event.Event.get():
470            processed = True
471            matched_any = False
472
473            if event.type == pygame.KEYDOWN:
474
475                if event.key not in self.__key_scancodes:
476                    self.__key_scancodes[event.key] = True
477
478                self.__key_scancodes[event.key] = True
479
480            if event.type == pygame.KEYUP:
481
482                if event.key not in self.__key_scancodes:
483                    self.__key_scancodes[event.key] = False
484
485                self.__key_scancodes[event.key] = False
486
487            for ecallback in self.__event_callbacks:
488
489                # First, always match the event type.
490                if event.type != ecallback.type:
491                    continue
492
493                # Then, filter key-specific callbacks.
494                if event.type == pygame.KEYDOWN:
495
496                    if (
497                        ecallback.data is not None
498                        and "key" in ecallback.data
499                        and event.key != ecallback.data["key"]
500                    ):
501                        continue
502
503                matched_any = True
504
505                res = ecallback(event)
506
507                if res == Frame.EventCallback.EXIT:
508                    self.raise_exit_flag()
509
510                if res == Frame.EventCallback.FAILURE:
511                    processed = False
512
513                if ecallback.putback:
514                    processed = False
515
516            if not processed or not matched_any:
517                putback_events.append(event)
518
519        for event in putback_events:
520            pygame.event.Event.post(event)
