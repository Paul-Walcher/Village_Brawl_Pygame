 80    def _check_max_zoom_values(self) -> bool:
 81        """
 82        Returns True if everything is alright,
 83        False if any value was invalid and had to be changed.
 84        """
 85
 86        valid: bool = True
 87
 88        if  self._max_zoom_in_value > Camera.ABSOLUTE_MAX_ZOOM_IN_VALUE - EPSILON\
 89            or self._max_zoom_in_value < 1.0 + EPSILON:
 90            #invalid
 91            self._max_zoom_in_value = Camera.STANDARD_MAX_ZOOM_IN_VALUE - 2 * EPSILON
 92            valid = False
 93
 94        if  self._max_zoom_out_value < Camera.ABSOLUTE_MAX_ZOOM_OUT_VALUE + EPSILON\
 95            or self._max_zoom_out_value > 1.0 - EPSILON:
 96            #invalid
 97            self._max_zoom_out_value = Camera.STANDARD_MAX_ZOOM_OUT_VALUE + 2 * EPSILON
 98            valid = False
 99
100        return valid

102    def _check_current_zoom_value(self) -> bool:
103        """
104        Returns True if everything is alright,
105        False if any value was invalid and had to be changed.
106        """
107
108        valid: bool = True
109
110        if self._current_zoom > (self._max_zoom_in_value - EPSILON):
111
112            self._current_zoom = (self._max_zoom_in_value - 2 * EPSILON) #not setting directly to max
113            valid = False

115        if self._current_zoom < (self._max_zoom_out_value + EPSILON):
116
117            self._current_zoom = (self._max_zoom_out_value + 2 * EPSILON) #not setting directly to max
118            valid = False
119
120        return valid

253    def render_surface(
254                        self,
255                        screen: pygame.Surface,
256                        surface: pygame.Surface
257                        ) -> None:
258        """
259        Applies all changes, and renders the surface onto the screen.
260        """
261
262        x: int = None
263        y: int = None
264        #x and y starting positions
265        x, y = self.frame_dim.x, self.frame_dim.y
266
267        #applying offsets
268        x += int(self._x_offset)
269        y += int(self._y_offset)
270
271        #scaling the given surface
272        if not fequal(self._current_zoom, 1.0):
273            #getting sizes
274            swidth: int = surface.get_width()
275            sheight: int = surface.get_height()
276
277            new_width: int = int(swidth * self._current_zoom)
278            new_height: int = int(sheight * self._current_zoom)
279
280            extra_width: int = (new_width - swidth)
281            extra_height: int = (new_height - sheight)
282
283            x -= extra_width // 2
284            y -= extra_height // 2
285
286            surface = pygame.transform.smoothscale(surface=surface, size=(new_width, new_height))
287
288        screen.blit(surface, (x, y))

290    def transform_point(self, point: point_t) -> point_t:
291        """
292        Transforms the given point from the usual
293        screen space into the camera space.
294        """
295
296        x: float = None
297        y: float = None
298
299        x, y = point
300
301        #positioning x and y relative to the own x and y
302        x -= self.frame_dim.x
303        y -= self.frame_dim.y
304
305        #now applying the offsets
306        x -= self._x_offset
307        y -= self._y_offset
308
309        #applying the zoom
310        swidth: int = self.frame_dim.width
311        sheight: int = self.frame_dim.height
312
313        new_width: float = swidth * self._current_zoom
314        new_height: float = sheight * self._current_zoom
315
316        extra_width: float = (new_width - swidth)
317        extra_height: float = (new_height - sheight)
318
319        x += extra_width / 2
320        y += extra_height / 2
321
322        x /= self._current_zoom
323        y /= self._current_zoom
324
325
326        return (int(x), int(y))
327
328    def get_mouse_position(self) -> point_t:
329        """
330        Gets the actual mouse position in camera space,
331        using the zoom and offset.
332        """
333
334        return (self.transform_point(pygame.mouse.get_pos()))
335
336    def viewport(self) -> pygame.Rect:
337        """
338        Returns the viewport, i.e.
339        the rectangle of what is currently rendered.
340        """
341        x: int = 0
342        y: int = 0
343
344        x, y = self.transform_point((self.frame_dim.x, self.frame_dim.y))
345        #getting width and height
346        width: int = int(self.frame_dim.width / self._current_zoom)
347        height: int = int(self.frame_dim.height / self._current_zoom)
348
349        return pygame.Rect(x, y, width, height)

254    def render_surface(
255                        self,
256                        screen: pygame.Surface,

258                        ) -> None:
259        """
260        Applies all changes, and renders the surface onto the screen.
261        """
262
263        x: int = None
264        y: int = None
265        #x and y starting positions
266        x, y = self.frame_dim.x, self.frame_dim.y

270        y += int(self._y_offset)
271
272        #scaling the given surface
273        if not fequal(self._current_zoom, 1.0):
274            #getting sizes

276            sheight: int = surface.get_height()
277

282            extra_height: int = (new_height - sheight)
283
284            x -= extra_width // 2
285            y -= extra_height // 2
286
287            surface = pygame.transform.smoothscale(surface=surface, size=(new_width, new_height))
288

def transform_point(self, point: point_t) -> point_t: 
291    def transform_point(self, point: point_t) -> point_t:
292        """
293        Transforms the given point from the usual
294        screen space into the camera space.
295        """
296
297        x: float = None
298        y: float = None
299
300        x, y = point
301
302        #positioning x and y relative to the own x and y
303        x -= self.frame_dim.x
304        y -= self.frame_dim.y
305
306        #now applying the offsets
307        x -= self._x_offset
308        y -= self._y_offset
309
310        #applying the zoom
311        swidth: int = self.frame_dim.width
312        sheight: int = self.frame_dim.height
313
314        new_width: float = swidth * self._current_zoom
315        new_height: float = sheight * self._current_zoom
316
317        extra_width: float = (new_width - swidth)
318        extra_height: float = (new_height - sheight)
319
320        x += extra_width / 2
321        y += extra_height / 2
322
323        x /= self._current_zoom
324        y /= self._current_zoom
325
326
327        return (int(x), int(y))
